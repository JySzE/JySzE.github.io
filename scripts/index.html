$ErrorActionPreference = 'Stop' $baseUrl = 'https://jysze.github.io' $manifestUrl = "$baseUrl/manifest.json" function ConvertTo-Text { param([Parameter(Mandatory = $true)] $Response) if ($Response -is [string]) { return [string]$Response } if ($Response -is [byte[]]) { return [System.Text.Encoding]::UTF8.GetString($Response) } if ($null -ne $Response.RawContentStream) { $Response.RawContentStream.Position = 0 $reader = [System.IO.StreamReader]::new( $Response.RawContentStream, [System.Text.UTF8Encoding]::new($false), $true ) try { return $reader.ReadToEnd() } finally { $reader.Dispose() } } throw 'The server response could not be converted to text.' } function Select-MenuItem { param( [Parameter(Mandatory = $true)][object[]]$Items, [string]$Title = 'Select a script (Up/Down to move, Enter to run, Esc to cancel)' ) $selected = 0 $width = [Math]::Max(20, [Console]::WindowWidth - 1) $startTop = 0 $firstDraw = $true $oldCursor = [Console]::CursorVisible Write-Host $Title -ForegroundColor Cyan Write-Host '' [Console]::CursorVisible = $false try { while ($true) { if (-not $firstDraw) { [Console]::SetCursorPosition(0, $startTop) } for ($i = 0; $i -lt $Items.Count; $i++) { $text = " $($Items[$i].name) - $($Items[$i].description)" if ($text.Length -gt $width) { $text = $text.Substring(0, $width - 3) + '...' } $text = $text.PadRight($width) if ($i -eq $selected) { Write-Host $text -ForegroundColor Black -BackgroundColor Cyan } else { Write-Host $text } } if ($firstDraw) { $startTop = [Console]::CursorTop - $Items.Count $firstDraw = $false } $key = [Console]::ReadKey($true) switch ($key.Key) { 'UpArrow' { $selected = ($selected - 1 + $Items.Count) % $Items.Count } 'DownArrow' { $selected = ($selected + 1) % $Items.Count } 'Enter' { return $Items[$selected] } 'Escape' { return $null } } } } finally { [Console]::CursorVisible = $oldCursor } } $manifestResponse = Invoke-WebRequest -Uri $manifestUrl -UseBasicParsing -ErrorAction Stop $manifestJson = ConvertTo-Text $manifestResponse $manifest = $manifestJson | ConvertFrom-Json -ErrorAction Stop if ($null -eq $manifest -or $manifest.version -ne 1 -or $null -eq $manifest.scripts) { throw "The script manifest at $manifestUrl is invalid or uses an unsupported version." } $availableScripts = @($manifest.scripts) if ($availableScripts.Count -eq 0) { Write-Host 'No scripts are currently listed in the manifest.' return } foreach ($scriptEntry in $availableScripts) { if ([string]::IsNullOrWhiteSpace([string]$scriptEntry.name) -or [string]::IsNullOrWhiteSpace([string]$scriptEntry.description) -or [string]$scriptEntry.path -notmatch '^library/[A-Za-z0-9_-][A-Za-z0-9._-]*\.(ps1|bat|cmd|py)$') { throw 'The script manifest contains an invalid entry. Each entry needs a name, description, and a library path ending in .ps1, .bat, .cmd, or .py.' } } $selectedScript = Select-MenuItem -Items $availableScripts if ($null -eq $selectedScript) { Write-Host 'Cancelled.' return } Write-Host "Running '$($selectedScript.name)'..." -ForegroundColor Green $scriptUrl = "$baseUrl/$($selectedScript.path)" $response = Invoke-WebRequest -Uri $scriptUrl -UseBasicParsing -ErrorAction Stop $scriptText = ConvertTo-Text $response if ([string]::IsNullOrWhiteSpace($scriptText)) { throw "The script at $scriptUrl is empty." } $extension = [System.IO.Path]::GetExtension($selectedScript.path).ToLowerInvariant() $tempPath = Join-Path ([System.IO.Path]::GetTempPath()) ([Guid]::NewGuid().ToString() + $extension) try { $encoding = [System.Text.UTF8Encoding]::new($false) [System.IO.File]::WriteAllText($tempPath, $scriptText, $encoding) switch ($extension) { '.ps1' { Invoke-Expression $scriptText } { $_ -in '.bat', '.cmd' } { $commandProcessor = Get-Command -Name 'cmd.exe' -CommandType Application -ErrorAction Stop $process = Start-Process -FilePath $commandProcessor.Source ` -ArgumentList @('/d', '/c', "`"$tempPath`"") ` -NoNewWindow -Wait -PassThru if ($process.ExitCode -ne 0) { throw "The batch script exited with code $($process.ExitCode)." } } '.py' { $python = Get-Command -Name @('py', 'python', 'python3') ` -CommandType Application -ErrorAction SilentlyContinue | Select-Object -First 1 if ($null -eq $python) { throw 'Python was not found. Install Python and ensure py, python, or python3 is on PATH.' } if ($python.Name -match '^py(\.exe)?$') { & $python.Source -3 $tempPath } else { & $python.Source $tempPath } if ($LASTEXITCODE -ne 0) { throw "The Python script exited with code $LASTEXITCODE." } } } } finally { if (Test-Path -LiteralPath $tempPath) { Remove-Item -LiteralPath $tempPath -Force -ErrorAction Stop } }
